```typescript
import type { FastifyPluginAsync } from "fastify"
import { sql } from "../db/client.js"
import { valkey } from "../lib/valkey.js"
import { lifecycleQueue } from "../queues/lifecycle.queue.js"
import { logAuditEvent } from "../lib/audit-log.js"

const ERASURE_GRACE_PERIOD_MS = 7 * 24 * 60 * 60 * 1000 // 7 days

const erasureRoutes: FastifyPluginAsync = async (fastify) => {

  // Auth guard
  fastify.addHook("preHandler", async (request, reply) => {
    if (!request.userId) {
      return reply.status(401).send({ error: "Unauthorized" })
    }
  })

  // ── POST /api/me/request-erasure ──────────────────────────────────────
  // UER-01: User requests erasure of their personal data.
  // UER-04: Immediately invalidates ALL sessions via Valkey blocklist.
  fastify.post("/api/me/request-erasure", async (request, reply) => {
    const userId = request.userId

    // Check for existing pending erasure
    const existing = await sql`
      SELECT id FROM user_erasure_requests
      WHERE user_id = ${userId}::uuid
        AND status = 'pending'
      LIMIT 1
    `
    if (existing.length > 0) {
      return reply.status(409).send({ error: "You already have a pending erasure request" })
    }

    const now = new Date()
    const scheduledAt = new Date(now.getTime() + ERASURE_GRACE_PERIOD_MS)
    const jobId = `user-erase:${userId}`

    // Create erasure request row (workspace_id is NULL — erasure is user-level)
    const [erasureRequest] = await sql`
      INSERT INTO user_erasure_requests (user_id, scheduled_at, job_id)
      VALUES (${userId}::uuid, ${scheduledAt.toISOString()}::timestamptz, ${jobId})
      RETURNING id
    `

    // Enqueue delayed BullMQ job
    // workspace_id in job data is empty string since erasure is user-level
    await lifecycleQueue.add(
      "user-erasure",
      { type: "user-erasure", userId, workspaceId: "" },
      { delay: ERASURE_GRACE_PERIOD_MS, jobId }
    )

    // UER-04: IMMEDIATELY invalidate sessions in ALL workspaces
    const memberships = await sql<{ workspace_id: string }[]>`
      SELECT workspace_id FROM workspace_members
      WHERE user_id = ${userId}::uuid
        AND is_active = true
    `

    const pipeline = valkey.pipeline()
    for (const m of memberships) {
      // Session plugin checks: deactivated:{workspaceId}:{userId}
      // No TTL — stays until erasure completes or is cancelled
      pipeline.set(`deactivated:${m.workspace_id}:${userId}`, "1")
    }
    await pipeline.exec()

    await logAuditEvent("user", userId, "requested", {
      scheduled_at: scheduledAt.toISOString(),
      workspaces_blocked: memberships.length,
    })

    return reply.send({
      erasure_request_id: erasureRequest!.id,
      status: "pending",
      scheduled_at: scheduledAt.toISOString(),
    })
  })

  // ── POST /api/me/cancel-erasure ──────────────────────────────────────
  // UER-02: User cancels pending erasure during 7-day grace period.
  fastify.post("/api/me/cancel-erasure", async (request, reply) => {
    const userId = request.userId

    // Find pending erasure request
    const [erasureRequest] = await sql`
      SELECT id, job_id FROM user_erasure_requests
      WHERE user_id = ${userId}::uuid
        AND status = 'pending'
      ORDER BY requested_at DESC
      LIMIT 1
    `
    if (!erasureRequest) {
      return reply.status(404).send({ error: "No pending erasure request found" })
    }

    // Remove BullMQ job
    const jobId = erasureRequest.job_id as string | null
    if (jobId) {
      const job = await lifecycleQueue.getJob(jobId)
      if (job) {
        await job.remove()
      }
    }

    // Mark as cancelled
    await sql`
      UPDATE user_erasure_requests
      SET status = 'cancelled'
      WHERE id = ${erasureRequest.id}::uuid
    `

    // Remove Valkey blocklist entries — restore session access
    const memberships = await sql<{ workspace_id: string }[]>`
      SELECT workspace_id FROM workspace_members
      WHERE user_id = ${userId}::uuid
        AND is_active = true
    `

    if (memberships.length > 0) {
      const keys = memberships.map(m => `deactivated:${m.workspace_id}:${userId}`)
      await valkey.del(...keys)
    }

    await logAuditEvent("user", userId, "cancelled")

    return reply.send({ status: "cancelled" })
  })

  // ── GET /api/me/erasure-status ────────────────────────────────────────
  // Returns current erasure status for the user (useful for profile UI).
  fastify.get("/api/me/erasure-status", async (request, reply) => {
    const userId = request.userId

    const [erasureRequest] = await sql`
      SELECT id, status, requested_at, scheduled_at
      FROM user_erasure_requests
      WHERE user_id = ${userId}::uuid
        AND status IN ('pending', 'processing')
      ORDER BY requested_at DESC
      LIMIT 1
    `

    if (!erasureRequest) {
      return reply.send({ has_pending_erasure: false })
    }

    return reply.send({
      has_pending_erasure: true,
      erasure_request_id: erasureRequest.id,
      status: erasureRequest.status,
      requested_at: erasureRequest.requested_at,
      scheduled_at: erasureRequest.scheduled_at,
    })
  })
}

export default erasureRoutes
```

Key design decisions:
- Erasure routes live in their own file (not profile.ts) to keep concerns separated.
- Session invalidation happens via Valkey pipeline for all workspaces at once.
- No TTL on blocklist keys — they persist until erasure completes or is cancelled.
- workspace_id in user_erasure_requests is NULL (erasure is user-level per GM1).
- Added GET status endpoint for Phase 3 TRN-02 frontend use.
