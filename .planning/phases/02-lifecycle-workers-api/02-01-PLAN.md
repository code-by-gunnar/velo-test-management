```typescript
import type { FastifyPluginAsync } from "fastify"
import { sql } from "../db/client.js"
import { lifecycleQueue } from "../queues/lifecycle.queue.js"
import { logAuditEvent } from "../lib/audit-log.js"

const GRACE_PERIOD_MS = 30 * 24 * 60 * 60 * 1000 // 30 days

const lifecycleRoutes: FastifyPluginAsync = async (fastify) => {

  // Auth guard — all lifecycle routes require authentication
  fastify.addHook("preHandler", async (request, reply) => {
    if (!request.userId) {
      return reply.status(401).send({ error: "Unauthorized" })
    }
  })

  // Helper: verify user is admin of the given workspace
  async function requireAdmin(userId: string, workspaceId: string): Promise<boolean> {
    const rows = await sql`
      SELECT role FROM workspace_members
      WHERE workspace_id = ${workspaceId}::uuid
        AND user_id = ${userId}::uuid
        AND is_active = true
    `
    return rows.length > 0 && rows[0]?.role === "admin"
  }

  // ── POST /api/workspaces/:workspaceId/lifecycle/request-deletion ──────
  // WLC-01: Admin requests workspace deletion. Sets 30-day grace period.
  fastify.post<{
    Params: { workspaceId: string }
  }>("/api/workspaces/:workspaceId/lifecycle/request-deletion", async (request, reply) => {
    const { workspaceId } = request.params
    const userId = request.userId

    // Admin check
    const isAdmin = await requireAdmin(userId, workspaceId)
    if (!isAdmin) {
      return reply.status(403).send({ error: "Admin access required" })
    }

    // Check workspace exists and is not already pending deletion
    const [workspace] = await sql`
      SELECT id, deletion_status FROM workspaces WHERE id = ${workspaceId}::uuid
    `
    if (!workspace) {
      return reply.status(404).send({ error: "Workspace not found" })
    }
    if (workspace.deletion_status) {
      return reply.status(409).send({
        error: "Workspace already has a pending or active deletion",
        deletion_status: workspace.deletion_status,
      })
    }

    // Calculate scheduled deletion date
    const now = new Date()
    const scheduledAt = new Date(now.getTime() + GRACE_PERIOD_MS)
    const jobId = `ws-delete:${workspaceId}`

    // Update workspace with deletion metadata
    await sql`
      UPDATE workspaces SET
        deletion_requested_at = ${now.toISOString()}::timestamptz,
        deletion_scheduled_at = ${scheduledAt.toISOString()}::timestamptz,
        deletion_requested_by = ${userId}::uuid,
        deletion_job_id = ${jobId},
        deletion_status = 'pending_deletion',
        updated_at = NOW()
      WHERE id = ${workspaceId}::uuid
    `

    // Enqueue delayed BullMQ job
    await lifecycleQueue.add(
      "workspace-delete",
      { type: "workspace-delete", workspaceId, requestedBy: userId },
      { delay: GRACE_PERIOD_MS, jobId }
    )

    await logAuditEvent("workspace", workspaceId, "requested", {
      requested_by: userId,
      scheduled_at: scheduledAt.toISOString(),
    })

    return reply.send({
      deletion_status: "pending_deletion",
      deletion_requested_at: now.toISOString(),
      deletion_scheduled_at: scheduledAt.toISOString(),
    })
  })

  // ── POST /api/workspaces/:workspaceId/lifecycle/cancel-deletion ──────
  // WLC-02: Admin cancels pending workspace deletion during grace period.
  fastify.post<{
    Params: { workspaceId: string }
  }>("/api/workspaces/:workspaceId/lifecycle/cancel-deletion", async (request, reply) => {
    const { workspaceId } = request.params
    const userId = request.userId

    // Admin check
    const isAdmin = await requireAdmin(userId, workspaceId)
    if (!isAdmin) {
      return reply.status(403).send({ error: "Admin access required" })
    }

    // Check workspace is actually pending deletion
    const [workspace] = await sql`
      SELECT id, deletion_status, deletion_job_id FROM workspaces
      WHERE id = ${workspaceId}::uuid
    `
    if (!workspace) {
      return reply.status(404).send({ error: "Workspace not found" })
    }
    if (workspace.deletion_status !== "pending_deletion") {
      return reply.status(409).send({
        error: "Workspace is not pending deletion",
        deletion_status: workspace.deletion_status ?? null,
      })
    }

    // Remove the BullMQ delayed job
    const jobId = workspace.deletion_job_id as string | null
    if (jobId) {
      const job = await lifecycleQueue.getJob(jobId)
      if (job) {
        await job.remove()
      }
    }

    // Clear all deletion columns
    await sql`
      UPDATE workspaces SET
        deletion_requested_at = NULL,
        deletion_scheduled_at = NULL,
        deletion_requested_by = NULL,
        deletion_job_id = NULL,
        deletion_status = NULL,
        updated_at = NOW()
      WHERE id = ${workspaceId}::uuid
    `

    await logAuditEvent("workspace", workspaceId, "cancelled", {
      cancelled_by: userId,
    })

    return reply.send({ deletion_status: null })
  })

  // ── GET /api/workspaces/:workspaceId/lifecycle/status ────────────────
  // Returns current deletion status for the workspace (useful for UI).
  fastify.get<{
    Params: { workspaceId: string }
  }>("/api/workspaces/:workspaceId/lifecycle/status", async (request, reply) => {
    const { workspaceId } = request.params

    // Any workspace member can view status
    const member = await sql`
      SELECT id FROM workspace_members
      WHERE workspace_id = ${workspaceId}::uuid
        AND user_id = ${request.userId}::uuid
        AND is_active = true
    `
    if (member.length === 0) {
      return reply.status(403).send({ error: "Access denied" })
    }

    const [workspace] = await sql`
      SELECT deletion_status, deletion_requested_at, deletion_scheduled_at, deletion_requested_by
      FROM workspaces WHERE id = ${workspaceId}::uuid
    `
    if (!workspace) {
      return reply.status(404).send({ error: "Workspace not found" })
    }

    return reply.send({
      deletion_status: workspace.deletion_status ?? null,
      deletion_requested_at: workspace.deletion_requested_at ?? null,
      deletion_scheduled_at: workspace.deletion_scheduled_at ?? null,
      deletion_requested_by: workspace.deletion_requested_by ?? null,
    })
  })
}

export default lifecycleRoutes
```

Key design decisions:
- Direct `sql` queries (not `withWorkspace`) because workspace deletion columns are not RLS-scoped — they're workspace-level operations that need to work even for workspaces being deleted.
- Admin verification via workspace_members query (same pattern as PATCH /api/workspaces/:id/slug).
- Added a GET status endpoint for the frontend (Phase 3 TRN-02 will use it).
