**listR2Objects** — paginated listing by prefix:
```typescript
/**
 * List all object keys under a given prefix.
 * Handles pagination automatically (1000 keys per page).
 * Returns empty array if R2 is not configured.
 */
export async function listR2Objects(prefix: string): Promise<string[]> {
  if (!R2_BUCKET_NAME || !r2Enabled()) return []

  const client = getR2Client()
  const keys: string[] = []
  let continuationToken: string | undefined

  do {
    const response = await client.send(new ListObjectsV2Command({
      Bucket: R2_BUCKET_NAME,
      Prefix: prefix,
      ContinuationToken: continuationToken,
    }))

    if (response.Contents) {
      for (const obj of response.Contents) {
        if (obj.Key) keys.push(obj.Key)
      }
    }

    continuationToken = response.IsTruncated ? response.NextContinuationToken : undefined
  } while (continuationToken)

  return keys
}
```

**deleteR2Objects** — batch delete in 1000-key chunks:
```typescript
/**
 * Batch-delete R2 objects by key. Handles batching (max 1000 per request).
 * Returns count of keys submitted for deletion.
 * Returns 0 if R2 is not configured or keys array is empty.
 */
export async function deleteR2Objects(keys: string[]): Promise<number> {
  if (keys.length === 0) return 0
  if (!R2_BUCKET_NAME || !r2Enabled()) return 0

  const client = getR2Client()
  let deleted = 0

  for (let i = 0; i < keys.length; i += 1000) {
    const batch = keys.slice(i, i + 1000)
    await client.send(new DeleteObjectsCommand({
      Bucket: R2_BUCKET_NAME,
      Delete: {
        Objects: batch.map(Key => ({ Key })),
        Quiet: true,
      },
    }))
    deleted += batch.length
  }

  return deleted
}
```

**Step 2: Replace the workspace-delete stub** with:
```typescript
case "workspace-delete": {
  const { workspaceId, requestedBy } = job.data
  console.log(`[lifecycle-worker] workspace-delete: ${workspaceId}`)

  // Step 1: Atomic status claim — prevents concurrent execution (INF-02)
  const claimed = await sql`
    UPDATE workspaces
    SET deletion_status = 'processing'
    WHERE id = ${workspaceId}::uuid
      AND deletion_status = 'pending_deletion'
    RETURNING id
  `
  if (claimed.length === 0) {
    console.log(`[lifecycle-worker] workspace ${workspaceId} already processing or gone, skipping`)
    break
  }

  await logAuditEvent("workspace", workspaceId, "processing", { requestedBy })

  // Step 2: Collect R2 keys BEFORE cascade (WLC-04 — critical ordering)
  let r2KeysDeleted = 0
  if (r2Enabled()) {
    // 2a. Avatar keys — only for users who are EXCLUSIVELY in this workspace
    // (users in multiple workspaces keep their avatar)
    const singleWsUsers = await sql<{ avatar_url: string }[]>`
      SELECT u.avatar_url FROM users u
      WHERE u.avatar_url IS NOT NULL
        AND u.id IN (
          SELECT wm.user_id FROM workspace_members wm
          WHERE wm.workspace_id = ${workspaceId}::uuid
        )
        AND NOT EXISTS (
          SELECT 1 FROM workspace_members wm2
          WHERE wm2.user_id = u.id
            AND wm2.workspace_id != ${workspaceId}::uuid
            AND wm2.is_active = true
        )
    `
    const avatarKeys = singleWsUsers.map(u => u.avatar_url)

    // 2b. Ingestion keys — all CI payloads for this workspace
    const ingestionKeys = await listR2Objects(`ingestion/${workspaceId}/`)

    // 2c. Batch delete all collected R2 objects
    const allKeys = [...avatarKeys, ...ingestionKeys]
    if (allKeys.length > 0) {
      r2KeysDeleted = await deleteR2Objects(allKeys)
      console.log(`[lifecycle-worker] deleted ${r2KeysDeleted} R2 objects for workspace ${workspaceId}`)
    }
  }

  // Step 3: Clean up Valkey keys by workspace pattern
  try {
    const roleKeys = await valkey.keys(`member_role:${workspaceId}:*`)
    const deactKeys = await valkey.keys(`deactivated:${workspaceId}:*`)
    const allValkeyKeys = [...roleKeys, ...deactKeys]
    if (allValkeyKeys.length > 0) {
      await valkey.del(...allValkeyKeys)
      console.log(`[lifecycle-worker] cleaned ${allValkeyKeys.length} Valkey keys for workspace ${workspaceId}`)
    }
  } catch (err) {
    // Non-fatal — Valkey keys expire naturally
    console.error(`[lifecycle-worker] Valkey cleanup warning:`, (err as Error).message)
  }

  // Step 4: Hard-delete workspace row — CASCADE handles all child tables
  await sql`DELETE FROM workspaces WHERE id = ${workspaceId}::uuid`

  await logAuditEvent("workspace", workspaceId, "completed", {
    r2_objects_deleted: r2KeysDeleted,
  })

  console.log(`[lifecycle-worker] workspace ${workspaceId} deleted`)
  break
}
```

**Step 3: Replace the user-erasure stub** with:
```typescript
case "user-erasure": {
  const { userId } = job.data
  console.log(`[lifecycle-worker] user-erasure: ${userId}`)

  // Step 1: Atomic status claim on the erasure request (INF-02)
  const claimed = await sql`
    UPDATE user_erasure_requests
    SET status = 'processing'
    WHERE user_id = ${userId}::uuid
      AND status = 'pending'
    RETURNING id
  `
  if (claimed.length === 0) {
    console.log(`[lifecycle-worker] erasure for user ${userId} already processing or done, skipping`)
    break
  }

  const erasureRequestId = claimed[0]!.id as string
  await logAuditEvent("user", userId, "processing")

  // Step 2: Delete avatar from R2 (if exists)
  if (r2Enabled()) {
    const [user] = await sql<{ avatar_url: string | null }[]>`
      SELECT avatar_url FROM users WHERE id = ${userId}::uuid
    `
    if (user?.avatar_url) {
      await deleteR2Objects([user.avatar_url])
      console.log(`[lifecycle-worker] deleted avatar for user ${userId}`)
    }
  }

  // Step 3: Anonymize ALL PII fields (UER-03)
  // deleted-{uuid}@deleted.invalid frees the original email for re-registration (GC5)
  const anonEmail = `deleted-${erasureRequestId}@deleted.invalid`
  await sql`
    UPDATE users SET
      name = 'Deleted User',
      email = ${anonEmail},
      password_hash = NULL,
      avatar_url = NULL,
      pending_email = NULL,
      updated_at = NOW()
    WHERE id = ${userId}::uuid
  `
  // UER-05: User row is preserved (UPDATE not DELETE).
  // created_by UUIDs in test_cases, run_items, etc. still reference
  // this row — JOINs now show "Deleted User" instead of real name.

  // Step 4: Mark erasure request as completed
  await sql`
    UPDATE user_erasure_requests
    SET status = 'completed', completed_at = NOW()
    WHERE id = ${erasureRequestId}::uuid
  `

  // Step 5: Clean up Valkey blocklist keys
  try {
    const blockKeys = await valkey.keys(`deactivated:*:${userId}`)
    if (blockKeys.length > 0) {
      await valkey.del(...blockKeys)
    }
  } catch (err) {
    console.error(`[lifecycle-worker] Valkey cleanup warning:`, (err as Error).message)
  }

  await logAuditEvent("user", userId, "completed", {
    erasure_request_id: erasureRequestId,
  })

  console.log(`[lifecycle-worker] user ${userId} anonymized`)
  break
}
```
