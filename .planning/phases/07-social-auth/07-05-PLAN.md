<!-- signIn() from next-auth/react -->

```typescript
import { signIn } from "next-auth/react"

// Triggers OAuth redirect to provider. callbackUrl is where user lands after auth.
signIn("google", { callbackUrl: "/onboarding" })
signIn("github", { callbackUrl: "/onboarding" })
```

The callbackUrl should be `/onboarding` since OAuth users may not have a workspace yet (JIT provisioned). The onboarding page handles redirect to app if workspace already exists.

<!-- OAuth error codes from signIn callback (07-04) -->

The signIn callback in auth.ts redirects to `/login?error=<code>` for these cases:
- `no_email` — provider didn't return an email address
- `oauth_error` — generic fetch failure calling Fastify backend
- `unverified_email` — existing account with unverified email
- `provider_conflict` — account already linked to a different provider

<!-- Auth.js internal error types (pages.error redirect) -->

Auth.js redirects to `pages.error` with `?error=<Type>` for:
- `Configuration` — server misconfiguration (missing env vars, wrong callback URL)
- `AccessDenied` — user denied consent on provider screen
- `Verification` — token/CSRF verification failed
- `OAuthSignin` — error starting OAuth flow
- `OAuthCallback` — error on callback from provider
- `OAuthCreateAccount` — error creating account
- `OAuthAccountNotLinked` — account exists with different provider (we handle this in signIn callback, but Auth.js can also throw it)
- `Default` — generic fallback

<!-- Design system tokens (from CLAUDE.md and tailwind.config.ts) -->

- Button size `lg` on auth pages (standalone pages, per design system rules)
- Card with `padding="lg"` for auth pages
- `bg-mist` page background
- `text-gray-500` for secondary text
- `text-primary hover:underline` for links
- `bg-fail-bg text-fail-text` for error alerts

interface SocialAuthButtonsProps {
  /** Where to redirect after successful auth */
  callbackUrl?: string
}

export function SocialAuthButtons({ callbackUrl = "/onboarding" }: SocialAuthButtonsProps) {
  const [loading, setLoading] = useState<"google" | "github" | null>(null)

  const handleSignIn = (provider: "google" | "github") => {
    setLoading(provider)
    signIn(provider, { callbackUrl })
  }

  return (
    <div className="flex flex-col gap-3">
      <button
        type="button"
        onClick={() => handleSignIn("google")}
        disabled={loading !== null}
        className="flex h-11 w-full items-center justify-center gap-2.5 rounded-md border border-gray-200 bg-white text-sm font-medium text-gray-700 transition-colors hover:bg-gray-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary focus-visible:ring-offset-2 disabled:opacity-40 disabled:pointer-events-none"

**Step 1: Create `apps/web/src/pages/auth/error.tsx`:**

```typescript
import Link from "next/link"
import Image from "next/image"
import { useRouter } from "next/router"
import { Card } from "@/components/ui"

/**
 * Maps Auth.js internal error types to human-readable messages.
 * These are errors thrown by Auth.js itself (not our signIn callback).
 * See: https://authjs.dev/guides/pages/error
 */
const errorMessages: Record<string, { title: string; message: string }> = {
  Configuration: {
    title: "Server configuration error",
    message: "There's a problem with the authentication setup. Please contact support if this persists.",
  },
  AccessDenied: {
    title: "Access denied",
    message: "You declined the sign-in request. If this was a mistake, try signing in again.",
  },
  Verification: {
    title: "Verification failed",
    message: "The sign-in link has expired or has already been used. Please request a new one.",
  },
  OAuthSignin: {
    title: "Could not start sign-in",
    message: "Something went wrong connecting to the sign-in provider. Please try again.",
  },
  OAuthCallback: {
    title: "Sign-in interrupted",
    message: "The sign-in provider returned an error. Please try again.",
  },
  OAuthCreateAccount: {
    title: "Could not create account",
    message: "There was a problem creating your account. Please try a different sign-in method.",
  },
  OAuthAccountNotLinked: {
    title: "Account already exists",
    message: "An account with this email already uses a different sign-in method. Please sign in with your original method.",
  },
  Default: {
    title: "Something went wrong",
    message: "An unexpected error occurred during sign-in. Please try again.",
  },
}

export default function AuthErrorPage() {
  const router = useRouter()
  const errorType = typeof router.query.error === "string" ? router.query.error : "Default"
  const { title, message } = errorMessages[errorType] ?? errorMessages.Default

  return (
    <div className="flex min-h-screen items-center justify-center bg-mist p-4">
      <div className="w-full max-w-md">
        <div className="mb-8 flex justify-center">
          <Image src="/velo-mark-cobalt.svg" alt="Velo" width={48} height={48} />
        </div>
        <Card padding="lg">
          <h1 className="text-xl font-semibold text-gray-900 mb-1">{title}</h1>
          <p className="text-sm text-gray-500 mb-6">{message}</p>

          <Link
            href="/login"
            className="inline-flex h-11 w-full items-center justify-center rounded-md bg-primary text-white text-base font-medium transition-colors hover:bg-primary-hover focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary focus-visible:ring-offset-2"
          >
            Back to sign in
          </Link>
        </Card>
      </div>
    </div>
  )
}
```

Design notes:
- Same layout as login/signup (centered card, Velo logo, bg-mist)
- Calm, professional tone — no warning icons, no red backgrounds
- Title + description pattern (same as login page)
- Single CTA: "Back to sign in" as a link styled like a primary lg button
- Handles unknown error types gracefully (falls back to Default)
