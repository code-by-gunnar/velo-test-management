```typescript
import Papa from "papaparse"
import ExcelJS from "exceljs"

// (Install papaparse in apps/api — it's in apps/web but we need server-side too)
// pnpm --filter @velo/api add papaparse @types/papaparse

export interface ParsedStep { action: string; expected_result: string }
export interface TestCaseImport {
  title: string
  preconditions?: string
  priority?: "critical" | "high" | "medium" | "low"
  steps: ParsedStep[]
}

// Auto-detect column mapping from header row
function detectColumns(headers: string[]): {
  title?: number; action?: number; expected?: number; preconditions?: number; priority?: number
} {
  const mapping: Record<string, number> = {}
  headers.forEach((h, i) => {
    const lower = h.toLowerCase().trim()
    if (["title", "test case", "name", "test name"].includes(lower)) mapping.title = i
    if (["action", "step", "step description", "description"].includes(lower)) mapping.action = i
    if (["expected", "expected result", "expected results"].includes(lower)) mapping.expected = i
    if (["preconditions", "precondition", "prerequisites"].includes(lower)) mapping.preconditions = i
    if (["priority", "severity"].includes(lower)) mapping.priority = i
  })
  return mapping
}

function normalizePriority(val: string): "critical"|"high"|"medium"|"low" | undefined {
  const v = val.toLowerCase().trim()
  if (["critical", "high", "medium", "low"].includes(v)) return v as any
  return undefined
}

// Group multi-row format (same title = same case, each row is a step)
function groupMultiRow(rows: string[][], cols: ReturnType<typeof detectColumns>): TestCaseImport[] {
  const cases = new Map<string, TestCaseImport>()
  for (const row of rows) {
    const title = row[cols.title!]?.trim()
    if (!title) continue
    if (!cases.has(title)) {
      cases.set(title, {
        title,
        preconditions: cols.preconditions !== undefined ? row[cols.preconditions]?.trim() || undefined : undefined,
        priority: cols.priority !== undefined ? normalizePriority(row[cols.priority] ?? "") : undefined,
        steps: [],
      })
    }
    const action = cols.action !== undefined ? row[cols.action]?.trim() : ""
    const expected = cols.expected !== undefined ? row[cols.expected]?.trim() : ""
    if (action) {
      cases.get(title)!.steps.push({ action, expected_result: expected || "" })
    }
  }
  return Array.from(cases.values())
}

export async function parseImportBuffer(buffer: Buffer, filename: string): Promise<TestCaseImport[]> {
  const lower = filename.toLowerCase()

  if (lower.endsWith(".csv")) {
    const text = buffer.toString("utf-8")
    const result = Papa.parse<string[]>(text, { skipEmptyLines: true })
    const [headerRow, ...dataRows] = result.data
    const cols = detectColumns(headerRow)
    if (cols.title === undefined) throw new Error("Missing required column: title")
    return groupMultiRow(dataRows, cols)
  }

  if (lower.endsWith(".xlsx") || lower.endsWith(".xls")) {
    const wb = new ExcelJS.Workbook()
    await wb.xlsx.load(buffer)
    const ws = wb.worksheets[0]
    const rows: string[][] = []
    ws.eachRow((row) => {
      rows.push(row.values.slice(1) as string[])  // slice(1) removes ExcelJS's 1-indexed offset
    })
    const [headerRow, ...dataRows] = rows
    const cols = detectColumns(headerRow.map(String))
    if (cols.title === undefined) throw new Error("Missing required column: title")
    return groupMultiRow(dataRows.map(r => r.map(String)), cols)
  }

  throw new Error("Unsupported file type")
}
```

```typescript
fastify.post("/api/workspaces/:workspaceId/projects/:projectId/cases/import", async (request, reply) => {
  const { workspaceId, projectId } = request.params as Record<string, string>
  if (request.workspaceId !== workspaceId) return reply.status(403).send()

  const data = await request.file()
  if (!data) return reply.status(400).send({ error: "No file" })
  const buffer = await data.toBuffer()

  let parsed: TestCaseImport[]
  try {
    parsed = await parseImportBuffer(buffer, data.filename)
  } catch (err: unknown) {
    return reply.status(422).send({ error: err instanceof Error ? err.message : "Parse error" })
  }

  await withWorkspace(workspaceId, async (tx) => {
    for (const tc of parsed) {
      const [existing] = await tx`
        SELECT COUNT(*)::int AS cnt FROM test_cases
        WHERE project_id = ${projectId} AND deleted_at IS NULL
      `
      if (existing.cnt >= 500) break  // Free tier limit — stop importing at cap

      const newCaseId = uuidv7()
      await tx`
        INSERT INTO test_cases (id, workspace_id, project_id, title, preconditions, priority, position, created_by)
        VALUES (${newCaseId}, ${workspaceId}, ${projectId}, ${tc.title},
                ${tc.preconditions ?? null}, ${tc.priority ?? "medium"},
                ${(existing.cnt + 1) * 1000}, ${request.userId})
      `
      for (let i = 0; i < tc.steps.length; i++) {
        await tx`
          INSERT INTO test_case_steps (id, test_case_id, step_order, action, expected_result)
          VALUES (${uuidv7()}, ${newCaseId}, ${(i + 1) * 1000}, ${tc.steps[i].action}, ${tc.steps[i].expected_result})
        `
      }
    }
  })

  return reply.status(201).send({ imported: parsed.length })
})
```
